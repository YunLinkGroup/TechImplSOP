```yaml
# ===========================
# KONG ENTERPRISE DOCKER-COMPOSE
# ===========================
# 此模板用于一键启动 Kong Gateway (Enterprise) + Postgres
#  1. Postgres：持久化元数据
#  2. kong-bootstrap：首次启动时自动建表
#  3. kong-cp：Kong 控制平面（兼代理流量）与 Kong Manager
#
# 快速开始：
#   export KONG_LICENSE_DATA='<your-license-json>'
#   docker compose up -d
#
# 常用地址：
#   Kong Manager : http://localhost:8002  (初始密码 handyshake)
#   Admin API    : http://localhost:8001
#   Proxy        : http://localhost:8000
# ===========================

volumes:
  kong_db_data: { }  # 具名卷，确保 Postgres 数据在容器重启/删除后仍存

networks:
  kong-ee-net: # 自定义桥接网络，Kong 与 Postgres 之间隔离通信
    driver: bridge

# 通用环境变量 used by Kong services (bootstrap and CP)
x-kong-config: &kong-env
  KONG_DATABASE: postgres             # 使用 Postgres 作为后端库
  KONG_PG_HOST: kong-ee-database      # Postgres 服务名，Docker DNS 自动解析
  KONG_PG_DATABASE: kong              # 数据库名
  KONG_PG_USER: kong                  # 用户名
  KONG_PG_PASSWORD: kong              # 密码

services:

  kong-ee-database:
    container_name: kong-ee-database
    image: postgres:latest           # Official Postgres image
    restart: on-failure              # Restart if the container fails
    volumes:
      - kong_db_data:/var/lib/postgresql/data  # 挂载卷持久化数据
    networks:
      - kong-ee-net                  # Connect to the shared Kong network
    environment:
      POSTGRES_USER: kong            # Set DB user inside the container
      POSTGRES_DB: kong              # Create this database on first run
      POSTGRES_PASSWORD: kong        # Set the password for the DB user
    healthcheck: # 在启动依赖服务之前，确保数据库已准备就绪
      test: [ "CMD", "pg_isready", "-U", "kong" ]
      interval: 5s
      timeout: 10s
      retries: 10
    ports:
      - '5432:5432'                  # 可选：在本地主机上公开Postgres进行调试

  kong-bootstrap:
    image: '${GW_IMAGE:-kong/kong:latest}'  # Kong Gateway image (default to latest version)
    container_name: kong-bootstrap
    networks:
      - kong-ee-net
    depends_on:
      kong-ee-database:
        condition: service_healthy   # Wait until Postgres is up and healthy
    restart: on-failure
    environment:
      <<: *kong-env                 # Reuse environment config from x-kong-config
      KONG_PASSWORD: handyshake    # Admin GUI password (required for RBAC)
    command: kong migrations up  # 运行数据库迁移以初始化 Kong schema

  kong-cp:
    image: '${GW_IMAGE:-kong/kong:latest}'  # Main Kong Gateway Control Plane (default to latest version)
    container_name: kong-cp
    restart: on-failure
    networks:
      - kong-ee-net
    environment:
      <<: *kong-env
      KONG_ADMIN_LISTEN: 0.0.0.0:8001, 0.0.0.0:8444 ssl  # Admin API on HTTP + HTTPS
      KONG_ADMIN_GUI_LISTEN: 0.0.0.0:8002, 0.0.0.0:8445 ssl  # Kong Manager on HTTP + HTTPS
      KONG_ADMIN_GUI_API_URL: http://110.42.60.147:8001
      KONG_ADMIN_GUI_URL: http://110.42.60.147:8002  # URL for GUI links
      KONG_PASSWORD: handyshake  # Required for logging in to Kong Manager (RBAC)
    depends_on:
      kong-bootstrap:
        condition: service_completed_successfully  # Start only after bootstrap has succeeded
    ports:
      - "8000:8000"  # Proxy HTTP
      - "8443:8443"  # Proxy HTTPS
      - "8001:8001"  # Admin API HTTP
      - "8444:8444"  # Admin API HTTPS
      - "8002:8002"  # Kong Manager HTTP
      - "8445:8445"  # Kong Manager HTTPS
```

mini_http.py

```python
#!/usr/bin/env python3
"""
mini_http.py
一个专门给 Kong Gateway 做调研用的“靶子”服务。
端口：8080
路由：
  GET  /           → 200 OK + 简单 JSON
  GET  /status/200 → 返回 200（可改数字测不同状态码）
  POST /echo        → 把请求体原样回写
  
curl http://127.0.0.1:8080
curl http://127.0.0.1:8080/status/201
curl -d 'Kong rocks' http://127.0.0.1:8080/echo
"""
import json
from http.server import HTTPServer, BaseHTTPRequestHandler

class MiniHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/":
            self.send(200, {"message": "hello from mini_http", "route": "/"})
        elif self.path.startswith("/status/"):
            try:
                code = int(self.path.split("/")[-1])
            except ValueError:
                code = 500
            self.send(code, {"status": code})
        else:
            self.send(404, {"error": "not found"})

    def do_POST(self):
        if self.path == "/echo":
            length = int(self.headers.get("Content-Length", 0))
            body = self.rfile.read(length).decode()
            self.send(200, {"echo": body})
        else:
            self.send(404, {"error": "not found"})

    def send(self, code, payload):
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.end_headers()
        self.wfile.write(json.dumps(payload, ensure_ascii=False).encode("utf-8"))

if __name__ == "__main__":
    addr = ("0.0.0.0", 9999)
    print(f"Starting mini_http on {addr[0]}:{addr[1]} ...")
    HTTPServer(addr, MiniHandler).serve_forever()
```
